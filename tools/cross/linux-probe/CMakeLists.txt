cmake_minimum_required(VERSION 3.24)
project(ArbiterLinuxProbe LANGUAGES C CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_CROSSCOMPILING)
    message(FATAL_ERROR "This probe must use the Linux x64 cross toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.12 REQUIRED COMPONENTS Core)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LINUX_SDK REQUIRED IMPORTED_TARGET libavformat libavcodec libavutil nss openssl dbus-1 x11)
add_executable(linux-probe main.cpp)
set_target_properties(linux-probe PROPERTIES AUTOMOC ON)
target_link_libraries(linux-probe PRIVATE Qt6::Core PkgConfig::LINUX_SDK)
qt_add_resources(linux-probe "probe" PREFIX "/" FILES main.cpp)
