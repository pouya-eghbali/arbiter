cmake_minimum_required(VERSION 3.24)
project(ArbiterWindowsProbe LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_CROSSCOMPILING OR NOT MSVC)
    message(FATAL_ERROR "This probe must use the Windows MSVC-ABI cross toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
add_executable(windows-probe main.cpp icon.rc)
target_compile_definitions(windows-probe PRIVATE UNICODE _UNICODE WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_link_libraries(windows-probe PRIVATE user32 bcrypt)
if(ARBITER_WINDOWS_OPENSSL)
    set(OPENSSL_ROOT_DIR "${ARBITER_WINDOWS_OPENSSL}")
    set(OPENSSL_USE_STATIC_LIBS TRUE)
    find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
    target_link_libraries(windows-probe PRIVATE OpenSSL::Crypto)
    target_compile_definitions(windows-probe PRIVATE ARBITER_PROBE_OPENSSL)
endif()
if(ARBITER_PROBE_QT)
    find_package(Qt6 6.12 REQUIRED COMPONENTS Core)
    add_executable(qt-probe qt.cpp)
    set_target_properties(qt-probe PROPERTIES AUTOMOC ON)
    target_link_libraries(qt-probe PRIVATE Qt6::Core)
    target_link_libraries(qt-probe PRIVATE windowsapp)
    qt_add_resources(qt-probe "probe" PREFIX "/" FILES main.cpp)
endif()
