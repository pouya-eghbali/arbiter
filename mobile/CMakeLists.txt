cmake_minimum_required(VERSION 3.24)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION.txt" ARBITER_VERSION LIMIT_COUNT 1)
    project(ArbiterMobile VERSION ${ARBITER_VERSION} LANGUAGES CXX)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ARBITER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
include("${ARBITER_ROOT}/cmake/Dependencies.cmake")
find_package(Qt6 ${ARBITER_QT_VERSION} REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network Svg Sql QuickDialogs2 QuickEffects QuickLayouts Multimedia)
qt_standard_project_setup(REQUIRES ${ARBITER_QT_SERIES})
# Prefer WebEngine wherever the selected Qt kit actually provides it. Phones
# use the native WebViews because their kits do not provide WebEngine.
find_package(Qt6 ${ARBITER_QT_VERSION} QUIET COMPONENTS WebEngineQuick WebChannel)
set(mobile_qml qml/MobileMain.qml qml/MobileSettings.qml qml/MobileColors.qml qml/MobileLanguagePicker.qml qml/PairingScanner.qml qml/MobileIconButton.qml qml/MobilePill.qml qml/MobileStartPage.qml qml/MobileThumbnail.qml qml/MobileTabsOverview.qml qml/MobileWorkspaces.qml qml/MobileSearchSheet.qml qml/MobileMoreSheet.qml qml/NativePane.qml qml/MobileButton.qml qml/MobileField.qml qml/MobileToggle.qml qml/MobileComboBox.qml qml/MobileSection.qml qml/MobileEmptyState.qml qml/MobileLogo.qml)
if(TARGET Qt6::WebEngineQuick)
    set(ARBITER_MOBILE_WEBENGINE ON)
    list(APPEND mobile_qml qml/WebEnginePane.qml qml/EngineProfiles.qml)
else()
    if(NOT IOS AND NOT ANDROID)
        message(FATAL_ERROR "The desktop mobile preview requires a Qt WebEngine kit")
    endif()
endif()
qt_add_executable(arbiter_mobile MANUAL_FINALIZATION
    src/main.cpp src/qrscanner.cpp src/qrscanner.h src/mobilemodel.cpp src/mobilemodel.h src/mobiletabs.h
    src/mobilevault.cpp src/mobilevault.h src/nativepage.cpp src/nativepage.h src/mobileplatform.h
    "${ARBITER_ROOT}/src/credentialsyncstate.cpp" "${ARBITER_ROOT}/src/credentialsyncstate.h")
include("${ARBITER_ROOT}/cmake/QRCodeScanner.cmake")
target_link_libraries(arbiter_mobile PRIVATE ZXing::ZXing)
target_include_directories(arbiter_mobile PRIVATE src "${ARBITER_ROOT}/src")
target_link_libraries(arbiter_mobile PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Network Qt6::Svg Qt6::Sql Qt6::QuickDialogs2 Qt6::QuickEffects Qt6::QuickLayouts Qt6::Multimedia)
target_compile_definitions(arbiter_mobile PRIVATE ARBITER_VERSION="${ARBITER_VERSION}")
if(ARBITER_MOBILE_WEBENGINE)
    target_sources(arbiter_mobile PRIVATE src/mobileinterceptor.h)
    target_compile_definitions(arbiter_mobile PRIVATE ARBITER_MOBILE_WEBENGINE)
    target_link_libraries(arbiter_mobile PRIVATE Qt6::WebEngineQuick Qt6::WebChannel)
endif()
if(NOT IOS)
    include("${ARBITER_ROOT}/cmake/AdBlock.cmake")
    target_link_libraries(arbiter_mobile PRIVATE arbiter_adblock_engine)
endif()
if(IOS)
    # Camera capture uses Apple's native backend; no FFmpeg media stack is bundled.
    qt_import_plugins(arbiter_mobile EXCLUDE Qt6::QFFmpegMediaPlugin INCLUDE Qt6::QDarwinMediaPlugin)
    enable_language(OBJCXX)
    enable_language(Swift)
    # Qt 6.12 exports a Clang-only root include flag. Xcode cannot separate
    # transitive flags for CMake's mixed Swift/C++ force-link compilation.
    # Module targets already export the same include directory as a system path.
    get_target_property(platform_options Qt6::Platform INTERFACE_COMPILE_OPTIONS)
    if(platform_options)
        list(FILTER platform_options EXCLUDE REGEX "idirafter")
        set_property(TARGET Qt6::Platform PROPERTY INTERFACE_COMPILE_OPTIONS "${platform_options}")
    endif()
    set_target_properties(arbiter_mobile PROPERTIES Swift_LANGUAGE_VERSION 5)
    option(ARBITER_IOS_DEVICE_TESTING "Omit the unapproved default-browser entitlement for local testing" OFF)
    set(ARBITER_APPLE_TEAM "" CACHE STRING "Apple development team for device provisioning")
    add_library(arbiter_mobile_native STATIC platform/ios/NativeVault.swift platform/ios/CredentialVaultShared.swift)
    target_link_libraries(arbiter_mobile_native PRIVATE "-framework CryptoKit" "-framework Security" "-framework AuthenticationServices" "-framework LocalAuthentication")
    set_target_properties(arbiter_mobile_native PROPERTIES Swift_LANGUAGE_VERSION 5 XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 18.0)
    target_sources(arbiter_mobile PRIVATE src/mobilecrypto.cpp platform/ios/mobileplatform_ios.mm)
    set_source_files_properties(platform/ios/mobileplatform_ios.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(arbiter_mobile PRIVATE arbiter_mobile_native "-framework UIKit" "-framework WebKit" "-framework BackgroundTasks")
    add_executable(arbiter_credentials MACOSX_BUNDLE platform/ios/CredentialProviderViewController.swift platform/ios/CredentialVaultShared.swift)
    target_link_libraries(arbiter_credentials PRIVATE "-framework UIKit" "-framework AuthenticationServices" "-framework CryptoKit" "-framework Security" "-framework LocalAuthentication")
    set_target_properties(arbiter_credentials PROPERTIES
        OUTPUT_NAME ArbiterCredentials MACOSX_BUNDLE_GUI_IDENTIFIER com.arbiter.mobile.CredentialProvider
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/Provider-Info.plist"
        BUNDLE_EXTENSION appex XCODE_PRODUCT_TYPE com.apple.product-type.app-extension
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER com.arbiter.mobile.CredentialProvider
        XCODE_ATTRIBUTE_APPLICATION_EXTENSION_API_ONLY YES Swift_LANGUAGE_VERSION 5
        XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/Provider.entitlements"
        XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 18.0)
    set(entitlements "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/App.entitlements")
    if(ARBITER_IOS_DEVICE_TESTING)
        set(entitlements "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/DeviceTesting.entitlements")
        target_compile_definitions(arbiter_mobile PRIVATE ARBITER_IOS_DEVICE_TESTING)
    endif()
    set_target_properties(arbiter_mobile PROPERTIES
        MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER com.arbiter.mobile
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/App-Info.plist"
        XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER com.arbiter.mobile
        XCODE_ATTRIBUTE_CODE_SIGN_ENTITLEMENTS "${entitlements}"
        XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 18.0
        XCODE_EMBED_APP_EXTENSIONS arbiter_credentials
        XCODE_EMBED_APP_EXTENSIONS_CODE_SIGN_ON_COPY YES)
    if(ARBITER_APPLE_TEAM)
        foreach(t arbiter_mobile arbiter_credentials arbiter_mobile_native)
            set_target_properties(${t} PROPERTIES XCODE_ATTRIBUTE_DEVELOPMENT_TEAM "${ARBITER_APPLE_TEAM}" XCODE_ATTRIBUTE_CODE_SIGN_STYLE Automatic)
        endforeach()
    endif()
    set_source_files_properties(platform/ios/Assets.xcassets PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
    target_sources(arbiter_mobile PRIVATE platform/ios/Assets.xcassets)
    set_target_properties(arbiter_mobile PROPERTIES XCODE_ATTRIBUTE_ASSETCATALOG_COMPILER_APPICON_NAME AppIcon)
    foreach(code en de fr it es pt ja zh)
        foreach(owner App Provider)
            set(file "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/${owner}Localization/${code}.lproj/InfoPlist.strings")
            set_source_files_properties("${file}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${code}.lproj")
            if(owner STREQUAL "App")
                target_sources(arbiter_mobile PRIVATE "${file}")
            else()
                target_sources(arbiter_credentials PRIVATE "${file}")
            endif()
        endforeach()
    endforeach()
    file(GLOB translations "${CMAKE_CURRENT_SOURCE_DIR}/assets/i18n/*.json")
    set_source_files_properties(${translations} PROPERTIES MACOSX_PACKAGE_LOCATION Resources/Translations)
    target_sources(arbiter_mobile PRIVATE ${translations})
    target_sources(arbiter_credentials PRIVATE ${translations})
elseif(ANDROID)
    qt_import_plugins(arbiter_mobile EXCLUDE Qt6::QFFmpegMediaPlugin INCLUDE Qt6::QAndroidMediaPlugin)
    qt_add_library(arbiter_sync_core SHARED platform/android/sync_core.cpp "${ARBITER_ROOT}/src/credentialsyncstate.cpp")
    target_include_directories(arbiter_sync_core PRIVATE "${ARBITER_ROOT}/src")
    target_link_libraries(arbiter_sync_core PRIVATE Qt6::Core)
    set_property(TARGET arbiter_mobile APPEND PROPERTY QT_ANDROID_EXTRA_LIBS "$<TARGET_FILE:arbiter_sync_core>")
    add_dependencies(arbiter_mobile arbiter_sync_core)
    target_sources(arbiter_mobile PRIVATE src/mobilecrypto.cpp platform/android/mobileplatform_android.cpp)
    set(android_package "${CMAKE_CURRENT_BINARY_DIR}/android-package")
    configure_file("assets/browser/forms.js" "${android_package}/assets/browser/forms.js" COPYONLY)
    file(GLOB_RECURSE package_files CONFIGURE_DEPENDS RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/platform/android" "${CMAKE_CURRENT_SOURCE_DIR}/platform/android/*")
    foreach(file ${package_files})
        configure_file("platform/android/${file}" "${android_package}/${file}" COPYONLY)
    endforeach()
    foreach(code en de fr it es pt ja zh)
        configure_file("assets/i18n/${code}.json" "${android_package}/assets/i18n/${code}.json" COPYONLY)
    endforeach()
    set_target_properties(arbiter_mobile PROPERTIES QT_ANDROID_PACKAGE_SOURCE_DIR "${android_package}" QT_ANDROID_APP_NAME Arbiter QT_ANDROID_APP_VERSION_CODE 2 QT_ANDROID_APP_VERSION_NAME "${ARBITER_VERSION}" QT_ANDROID_PACKAGE_NAME com.arbiter.mobile QT_ANDROID_MIN_SDK_VERSION 28 QT_ANDROID_TARGET_SDK_VERSION 37)
else()
    include("${ARBITER_ROOT}/cmake/QtKeychain.cmake")
    find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
    target_sources(arbiter_mobile PRIVATE src/mobileplatform_desktop.cpp "${ARBITER_ROOT}/src/credentialsynccrypto.cpp")
    target_link_libraries(arbiter_mobile PRIVATE qt6keychain OpenSSL::Crypto)
    set_target_properties(arbiter_mobile PROPERTIES MACOSX_BUNDLE TRUE)
endif()
# Reuse desktop controls and design tokens, instead of keeping a second UI library.
set(shared_qml "${ARBITER_ROOT}/qml/Theme.qml" "${ARBITER_ROOT}/qml/GlassSurface.qml")
foreach(control UiButton UiIconButton UiIcon UiTextField UiCheckBox UiComboBox UiToolTip UiPopup)
    list(APPEND shared_qml "${ARBITER_ROOT}/qml/ui/${control}.qml")
endforeach()
list(APPEND shared_qml "${ARBITER_ROOT}/qml/ui/LucideIcons.js")
set_source_files_properties("${ARBITER_ROOT}/qml/Theme.qml" PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
foreach(file ${shared_qml} ${mobile_qml})
    get_filename_component(alias "${file}" NAME)
    set_source_files_properties("${file}" PROPERTIES QT_RESOURCE_ALIAS "${alias}")
endforeach()
qt_add_qml_module(arbiter_mobile URI Arbiter VERSION 1.0 QML_FILES ${mobile_qml} ${shared_qml})
qt_add_resources(arbiter_mobile mobile_translations PREFIX /i18n BASE assets/i18n FILES assets/i18n/en.json assets/i18n/de.json assets/i18n/fr.json assets/i18n/it.json assets/i18n/es.json assets/i18n/pt.json assets/i18n/ja.json assets/i18n/zh.json)
qt_add_resources(arbiter_mobile mobile_form_script PREFIX /browser BASE assets/browser FILES assets/browser/forms.js)
qt_add_resources(arbiter_mobile mobile_lists PREFIX /adblock BASE "${ARBITER_ROOT}/assets/adblock" FILES "${ARBITER_ROOT}/assets/adblock/easylist.txt" "${ARBITER_ROOT}/assets/adblock/easyprivacy.txt")
qt_add_resources(arbiter_mobile mobile_webkit PREFIX /adblock BASE assets/adblock FILES assets/adblock/webkit.json)
qt_add_resources(arbiter_mobile mobile_brand PREFIX /brand BASE "${ARBITER_ROOT}/assets/icon/Arbiter.icon/Assets" FILES "${ARBITER_ROOT}/assets/icon/Arbiter.icon/Assets/valkyrie-source.png")
qt_add_resources(arbiter_mobile mobile_licenses PREFIX /licenses BASE "${ARBITER_ROOT}" FILES "${ARBITER_ROOT}/LICENSE" "${ARBITER_ROOT}/licenses/Lucide.txt" "${ARBITER_ROOT}/licenses/ZXingCpp.txt" "${ARBITER_ROOT}/licenses/AdBlock.txt" "${ARBITER_ROOT}/licenses/AdBlockDependencies.txt" "${ARBITER_ROOT}/licenses/EasyList.txt")
option(ARBITER_MOBILE_DESIGN_PREVIEW "Enable isolated desktop design-fixture rendering" OFF)
if(ARBITER_MOBILE_DESIGN_PREVIEW)
    if(IOS OR ANDROID)
        message(FATAL_ERROR "Design fixtures are desktop-only and must not be included in phone builds")
    endif()
    find_package(Qt6 ${ARBITER_QT_VERSION} REQUIRED COMPONENTS Test)
    target_link_libraries(arbiter_mobile PRIVATE Qt6::Test)
    target_sources(arbiter_mobile PRIVATE src/designpreview.cpp src/designpreview.h)
    target_compile_definitions(arbiter_mobile PRIVATE ARBITER_MOBILE_DESIGN_PREVIEW)
endif()
qt_finalize_executable(arbiter_mobile)

option(ARBITER_MOBILE_TESTS "Build isolated host tests for the mobile core" OFF)
if(ARBITER_MOBILE_TESTS AND NOT IOS AND NOT ANDROID)
    enable_testing()
    find_package(Qt6 ${ARBITER_QT_VERSION} REQUIRED COMPONENTS Test)
    qt_add_executable(mobile_core_test tests/mobile_core_test.cpp tests/platform_test.cpp
        src/mobilemodel.cpp src/mobilemodel.h src/mobiletabs.h src/mobilevault.cpp src/mobilevault.h
        src/nativepage.cpp src/nativepage.h src/mobileplatform.h
        "${ARBITER_ROOT}/src/credentialsyncstate.cpp" "${ARBITER_ROOT}/src/credentialsynccrypto.cpp")
    target_include_directories(mobile_core_test PRIVATE src "${ARBITER_ROOT}/src")
    target_link_libraries(mobile_core_test PRIVATE Qt6::Test Qt6::Gui Qt6::Qml Qt6::Network OpenSSL::Crypto arbiter_adblock_engine)
    add_test(NAME mobile_core COMMAND mobile_core_test)
    set_tests_properties(mobile_core PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
