cmake_minimum_required(VERSION 3.24)
project(ArbiterTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
include(../cmake/ArbiterWebEngine.cmake)
find_package(Qt6 6.12 REQUIRED COMPONENTS Gui Widgets Test)
include(../cmake/QtKeychain.cmake)
enable_testing()
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
add_library(passkey_core STATIC ../src/passkeyvault.cpp ../src/passkeyvault.h ../src/passkeycrypto.cpp ../src/passkeycrypto.h
    ../src/secretstore.cpp ../src/secretstore.h)
target_link_libraries(passkey_core PUBLIC Qt6::Core OpenSSL::Crypto qt6keychain)
add_executable(passkeyvault_test passkeyvault_test.cpp)
target_link_libraries(passkeyvault_test PRIVATE passkey_core Qt6::Test Qt6::Widgets)
add_test(NAME passkeyvault COMMAND passkeyvault_test)
add_executable(keybindings_test keybindings_test.cpp ../src/keybindings.cpp ../src/keybindings.h)
target_link_libraries(keybindings_test PRIVATE Qt6::Gui Qt6::Test)
add_test(NAME keybindings COMMAND keybindings_test)
set_tests_properties(keybindings PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(nativeurl_test nativeurl_test.cpp)
target_link_libraries(nativeurl_test PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME nativeurl COMMAND nativeurl_test)
set_tests_properties(nativeurl PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(tabs_test tabs_test.cpp ../src/tabs.cpp ../src/tabs.h)
target_link_libraries(tabs_test PRIVATE Qt6::Test)
add_test(NAME tabs COMMAND tabs_test)

add_library(password_core STATIC ../src/passwordmanager.cpp ../src/passwordmanager.h
    ../src/secretstore.cpp ../src/secretstore.h)
target_link_libraries(password_core PUBLIC Qt6::Core qt6keychain)
add_executable(passwordmanager_test passwordmanager_test.cpp)
target_link_libraries(passwordmanager_test PRIVATE password_core Qt6::Widgets Qt6::Test)
add_test(NAME passwordmanager COMMAND passwordmanager_test)
if(NOT APPLE)
    set_tests_properties(passwordmanager PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()

if(ARBITER_WEBENGINE_TESTS)
    find_package(Qt6 6.12 REQUIRED COMPONENTS WebEngineWidgets WebChannel)
    add_executable(passwordforms_test passwordforms_test.cpp)
    qt_add_resources(passwordforms_test password_scripts PREFIX /passwords BASE ../scripts FILES ../scripts/passwords.js)
    target_link_libraries(passwordforms_test PRIVATE password_core Qt6::WebEngineWidgets Qt6::WebChannel Qt6::Test)
    arbiter_configure_webengine_runtime(passwordforms_test)
    add_test(NAME passwordforms COMMAND passwordforms_test)
    set_tests_properties(passwordforms PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(passkeyengine_test passkeyengine_test.cpp)
    target_link_libraries(passkeyengine_test PRIVATE passkey_core Qt6::WebEngineWidgets Qt6::Test)
    arbiter_configure_webengine_runtime(passkeyengine_test)
    add_test(NAME passkeyengine COMMAND passkeyengine_test)
    set_tests_properties(passkeyengine PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_executable(webauthncontroller_test webauthncontroller_test.cpp
        ../src/webauthncontroller.cpp ../src/webauthncontroller.h)
    target_link_libraries(webauthncontroller_test PRIVATE Qt6::WebEngineWidgets Qt6::Test)

    add_test(NAME webauthncontroller COMMAND webauthncontroller_test)
    set_tests_properties(webauthncontroller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
