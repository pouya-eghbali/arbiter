cmake_minimum_required(VERSION 3.24)
project(ArbiterTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Gui Widgets Test)
enable_testing()
add_executable(keybindings_test keybindings_test.cpp ../src/keybindings.cpp ../src/keybindings.h)
target_link_libraries(keybindings_test PRIVATE Qt6::Gui Qt6::Test)
add_test(NAME keybindings COMMAND keybindings_test)
set_tests_properties(keybindings PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_executable(nativeurl_test nativeurl_test.cpp)
target_link_libraries(nativeurl_test PRIVATE Qt6::Widgets Qt6::Test)
add_test(NAME nativeurl COMMAND nativeurl_test)
set_tests_properties(nativeurl PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
