find_package(Qt6 ${ARBITER_QT_SERIES} REQUIRED COMPONENTS Quick QuickControls2 WebEngineQuick Svg)
add_executable(desktopwindow_test desktopwindow_test.cpp desktopwindow_test.h ../../src/keybindings.cpp ../../src/keybindings.h)
target_link_libraries(desktopwindow_test PRIVATE arbiter_settings Qt6::Quick Qt6::QuickControls2 Qt6::WebEngineQuick Qt6::Svg Qt6::Test)
include(../../cmake/LinuxWindowFrame.cmake)
arbiter_add_linux_window_frame(desktopwindow_test)
arbiter_add_native_glass(desktopwindow_test)
set_source_files_properties(../../qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
set(window_qml ../../qml/WebFullScreen.qml ../../qml/Theme.qml ../../qml/GlassSurface.qml
    ../../qml/ui/UiButton.qml ../../qml/ui/UiIcon.qml ../../qml/ui/UiToolTip.qml ../../qml/ui/LucideIcons.js ../../qml/DesktopWindowFrame.js ../../qml/DesktopWindowOverlayClip.qml ../../qml/DesktopWindowSurface.qml ../../qml/DesktopSettingsShortcut.qml ../../qml/FramelessResizeEdges.qml)

    set_source_files_properties("${file}" PROPERTIES QT_RESOURCE_ALIAS "${name}")
endforeach()
qt_add_qml_module(desktopwindow_test URI Arbiter VERSION 1.0 NO_IMPORT_SCAN
    RESOURCE_PREFIX /desktop-window-test QML_FILES ${window_qml})
arbiter_configure_webengine_runtime(desktopwindow_test)
add_test(NAME desktopwindow COMMAND desktopwindow_test)
set_tests_properties(desktopwindow PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" LABELS webengine-runtime)
